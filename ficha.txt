<script>
// === LÓGICA DE AUTENTICACIÓN ===
const USER_VALIDO = "admin";
const PASS_VALIDO = "1234";

document.getElementById('login-form').addEventListener('submit', function(e) {
    e.preventDefault();
    const userIn = document.getElementById('login-username').value;
    const passIn = document.getElementById('login-password').value;
    const errorMsg = document.getElementById('error-msg');

    if (userIn === USER_VALIDO && passIn === PASS_VALIDO) {
        document.getElementById('login-screen').classList.add('oculto');
        document.getElementById('app-screen').classList.remove('oculto');
        document.getElementById('usuario').value = userIn;
        errorMsg.classList.add('oculto');
        
        actualizarContador();
        actualizarFicha();
    } else {
        errorMsg.classList.remove('oculto');
    }
});

function cerrarSesion() {
    document.getElementById('app-screen').classList.add('oculto');
    document.getElementById('login-screen').classList.remove('oculto');
    document.getElementById('login-username').value = '';
    document.getElementById('login-password').value = '';
}

// === MANEJO DE HISTORIAL Y ALMACENAMIENTO ===
function obtenerHistorial() {
    const historial = localStorage.getItem('historialFichas');
    return historial ? JSON.parse(historial) : [];
}

function actualizarContador() {
    const historial = obtenerHistorial();
    document.getElementById('contador-fichas').textContent = `Registros: ${historial.length}`;
}

function guardarFichaConsolidada() {
    const datosNuevos = obtenerDatos();
    
    if (!datosNuevos.fichaNo && !datosNuevos.nombres) {
        alert('⚠️ Ingresa al menos el número de ficha o los nombres antes de guardar.');
        return;
    }

    const historial = obtenerHistorial();
    historial.push(datosNuevos);
    localStorage.setItem('historialFichas', JSON.stringify(historial));
    
    actualizarContador();
    alert(`✅ Ficha No. ${datosNuevos.fichaNo || 'nueva'} guardada exitosamente. Total en sistema: ${historial.length}`);
}

// === LÓGICA DE FICHAS DE IDENTIFICACIÓN ===
let fotoBase64 = "";

document.getElementById('fotoInput').addEventListener('change', function(e) {
    const archivo = e.target.files[0];
    if (archivo) {
        const lector = new FileReader();
        lector.onload = function(e) {
            fotoBase64 = e.target.result;
            document.getElementById('vistaFoto').innerHTML = `<img src="${fotoBase64}" alt="Foto">`;
        };
        lector.readAsDataURL(archivo);
    }
});

function obtenerDatos() {
    const c = id => document.getElementById(id)?.value || '';
    return {
        nombres: c('nombres'),
        fichaNo: c('fichaNo'),
        fechaNac: c('fechaNac'),

function exportarExcel() {
    const historial = obtenerHistorial();
    
    if (historial.length === 0) {
        alert('⚠️ No hay fichas guardadas. Guarda al menos una ficha antes de exportar.');
        return;
    }

    // Remueve las imágenes base64 para evitar celdas pesadas o corruptas en Excel
    const datosExcel = historial.map(ficha => {
        const copia = { ...ficha };
        delete copia.foto;
        return copia;
    });

    const hoja = XLSX.utils.json_to_sheet(datosExcel);
    const libro = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(libro, hoja, "Fichas_Registradas");
    XLSX.writeFile(libro, `Base_Datos_Fichas_${new Date().toISOString().slice(0,10)}.xlsx`);
}

function crearRespaldo() {
    const historial = obtenerHistorial();
    
    if (historial.length === 0) {
        alert('⚠️ No hay fichas registradas para realizar un respaldo.');
        return;
    }

    const fecha = new Date().toISOString().slice(0,10);
    const nombreArchivo = `Respaldo_Fichas_Total_${fecha}.json`;

    const blob = new Blob([JSON.stringify(historial, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const enlace = document.createElement('a');
    enlace.href = url;
    enlace.download = nombreArchivo;
    enlace.click();
    URL.revokeObjectURL(url);
}

function restaurarRespaldo(evento) {
    const archivo = evento.target.files[0];
    if (!archivo) return;

    const lector = new FileReader();
    lector.onload = function(e) {
        try {
            const datosImportados = JSON.parse(e.target.result);
            
            // Permite importar tanto una lista de fichas como una ficha individual
            let historial = obtenerHistorial();
            if (Array.isArray(datosImportados)) {
                historial = historial.concat(datosImportados);
            } else {
                historial.push(datosImportados);
            }

            localStorage.setItem('historialFichas', JSON.stringify(historial));
            actualizarContador();
            alert('✅ Respaldo restaurado e integrado correctamente');
        } catch (err) {
            alert('❌ El archivo no tiene un formato JSON válido');
        }
    };
    lector.readAsText(archivo);
}
